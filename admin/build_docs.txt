use Dompdf\Dompdf;

use Parsedown;

// ==================================================================
// POST HANDLER — build the PDFs
// ==================================================================
$results = [];
$error   = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['build'])) {
    if (!hash_equals($_SESSION['csrf'], $_POST['csrf'] ?? '')) {
        $error = 'Invalid request.';
    } else {
        $docsDir = __DIR__ . '/../docs';

        if (!is_dir($docsDir)) {
            $error = "Docs folder not found at {$docsDir}";
        } else {
            // Optional: institution name for the cover header on each PDF
            $brand    = app_settings($pdo);
            $instName = $brand['institution_name'] ?? ($brand['system_name'] ?? 'Institution');
            $sysName  = $brand['system_name'] ?? 'SmartAttend';

            // Every .md file in the docs folder
            $files = glob($docsDir . '/*.md');
            sort($files);

            foreach ($files as $mdPath) {
                $baseName = basename($mdPath, '.md');
                $pdfName  = $baseName . '.pdf';
                $pdfPath  = $docsDir . '/' . $pdfName;

                try {
                    // 1) Read markdown
                    $markdown = file_get_contents($mdPath);
                    if ($markdown === false) {
                        throw new Exception("Could not read {$mdPath}");
                    }

                    // 2) Convert to HTML
                    $parsedown = new Parsedown();
                    $parsedown->setSafeMode(true);          // sanitize raw HTML in markdown
                    $parsedown->setBreaksEnabled(true);     // single newlines become <br>
                    $bodyHtml  = $parsedown->text($markdown);

                    // 3) Detect if this is a "one-pager" (quick start / overview)
                    // If so, use a more compact style.
                    $isCompact = preg_match('/QUICK_START|OVERVIEW/i', $baseName);

                    // 4) Wrap in a full HTML doc with CSS
                    $fullHtml = build_doc_html($bodyHtml, $instName, $sysName, $isCompact);

                    // 5) Render via Dompdf
                    $options = new Options();
                    $options->set('isRemoteEnabled', false);
                    $options->set('isHtml5ParserEnabled', true);
                    $options->set('defaultFont', 'DejaVu Sans');

                    $dompdf = new Dompdf($options);
                    $dompdf->loadHtml($fullHtml);
                    $dompdf->setPaper('A4', 'portrait');
                    $dompdf->render();

                    // 6) Save to disk
                    file_put_contents($pdfPath, $dompdf->output());

                    $results[] = [
                        'source'  => basename($mdPath),
                        'target'  => $pdfName,
                        'size'    => filesize($pdfPath),
                        'success' => true,
                    ];
                } catch (Exception $e) {
                    $results[] = [
                        'source'  => basename($mdPath),
                        'target'  => $pdfName,
                        'error'   => $e->getMessage(),
                        'success' => false,
                    ];
                }
            }
        }
    }
}

// ==================================================================
// HTML BUILDER
// ==================================================================
function build_doc_html(string $bodyHtml, string $instName, string $sysName, bool $compact): string {

    $h1 = $compact ? '15pt'  : '18pt';
    $h2 = $compact ? '12pt'  : '14pt';
    $h3 = $compact ? '10.5pt': '12pt';

    // Pre-escape header/footer values
    $instEsc = htmlspecialchars($instName, ENT_QUOTES, 'UTF-8');
    $sysEsc  = htmlspecialchars($sysName,  ENT_QUOTES, 'UTF-8');
    $generated = date('d M Y');

    return <<<HTML
<!DOCTYPE html>
<html><head><meta charset="utf-8">
<style>
    @page { margin: 40px 45px 55px 45px; }

    body {
        font-family: DejaVu Sans, sans-serif;
        font-size: {$fs};
        color: #1e293b;
        line-height: 1.55;
    }

    /* ---------- COVER HEADER ---------- */
    .doc-header {
        border-bottom: 3px double #1e293b;
        padding-bottom: 12px;
        margin-bottom: 24px;
    }
    .doc-header table { width: 100%; border-collapse: collapse; }
    .doc-header td { vertical-align: middle; padding: 0; }
    .doc-header .institution {
        font-size: 10pt;
        font-weight: bold;
        color: #0f172a;
        text-transform: uppercase;
        letter-spacing: 1px;
        margin: 0;
    }
    .doc-header .system {
        font-size: 8.5pt;
        color: #64748b;
        margin: 2px 0 0 0;
    }
    .doc-header .generated {
        font-size: 8pt;
        color: #94a3b8;
        text-align: right;
        margin: 0;
    }

    /* ---------- HEADINGS ---------- */
    h1 {
        font-size: {$h1};
        color: #0f172a;
        margin: 24px 0 10px 0;
        padding-bottom: 6px;
        border-bottom: 2px solid #1e293b;
        page-break-after: avoid;
    }
    h1:first-of-type { margin-top: 0; }

    h2 {
        font-size: {$h2};
        color: #0f172a;
        margin: 20px 0 8px 0;
        page-break-after: avoid;
    }

    h3 {
        font-size: {$h3};
        color: #1e293b;
        margin: 16px 0 6px 0;
        page-break-after: avoid;
    }

    /* ---------- BODY ---------- */
    p { margin: 8px 0; }

    ul, ol { margin: 8px 0; padding-left: 22px; }
    li { margin: 3px 0; }

    strong { color: #0f172a; }

    code {
        font-family: DejaVu Sans Mono, monospace;
        font-size: 0.9em;
        background: #f1f5f9;
        padding: 1px 5px;
        border-radius: 3px;
        color: #0f172a;
    }

    pre {
        background: #f8fafc;
        border-left: 4px solid #1a3ff0;
        padding: 10px 14px;
        border-radius: 4px;
        font-family: DejaVu Sans Mono, monospace;
        font-size: 0.85em;
        overflow-x: auto;
        page-break-inside: avoid;
    }
    pre code {
        background: none;
        padding: 0;
    }

    blockquote {
        border-left: 4px solid #cbd5e1;
        padding-left: 14px;
        margin: 12px 0;
        color: #64748b;
        font-style: italic;
    }

    hr {
        border: none;
        border-top: 1px solid #e2e8f0;
        margin: 20px 0;
    }

    a {
        color: #1a3ff0;
        text-decoration: none;
    }

    /* ---------- TABLES ---------- */
    table {
        width: 100%;
        border-collapse: collapse;
        margin: 14px 0;
        font-size: 0.92em;
        page-break-inside: avoid;
    }
    th {
        background: #1e293b;
        color: #fff;
        padding: 7px 9px;
        text-align: left;
        border: 1px solid #1e293b;
        font-weight: normal;
    }
    td {
        padding: 6px 9px;
        border: 1px solid #cbd5e1;
        vertical-align: top;
    }
    tr:nth-child(even) td { background: #f8fafc; }

    /* ---------- SCREENSHOTS OF ASCII BOXES ---------- */
    pre.ascii {
        background: #f8fafc;
        border: 1px solid #e2e8f0;
        border-left: 4px solid #1a3ff0;
        padding: 12px;
        font-size: 0.75em;
        line-height: 1.25;
        white-space: pre;
    }

    /* ---------- FOOTER ---------- */
    .doc-footer {
        position: fixed;
        bottom: -32px;
        left: 0; right: 0;
        text-align: center;
        font-size: 8pt;
        color: #94a3b8;
        border-top: 1px solid #e2e8f0;
        padding-top: 6px;
    }
    .doc-footer .page:after { content: counter(page); }

    /* ---------- PAGE BREAKS ---------- */
    .page-break { page-break-before: always; }
</style>
</head>
<body>

<div class="doc-header">
    <table>
        <tr>
            <td>
                <p class="institution">{$instEsc}</p>
                <p class="system">{$sysEsc} · Documentation</p>
            </td>
            <td style="width: 130px; text-align: right;">
                <p class="generated">Generated<br>{$generated}</p>
            </td>
        </tr>
    </table>
</div>

{$bodyHtml}

<div class="doc-footer">
    {$instEsc} · {$sysEsc} · <span class="page"></span>
</div>

</body></html>
HTML;
